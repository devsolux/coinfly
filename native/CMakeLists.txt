cmake_minimum_required(VERSION 3.26)
project(coinfly VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(APPLE)
  enable_language(OBJCXX)
  if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "The qualified Coinfly macOS target must explicitly set CMAKE_OSX_ARCHITECTURES=arm64")
  endif()
  add_library(coinfly_vault SHARED macos/keychain_bridge.mm)
  target_compile_options(coinfly_vault PRIVATE -fobjc-arc)
  target_link_libraries(coinfly_vault PRIVATE "-framework Foundation" "-framework Security")
  set(COINFLY_TEAM_ID "" CACHE STRING "Release signing team; empty keeps native access disabled")
  target_compile_definitions(coinfly_vault PRIVATE COINFLY_TEAM_ID="${COINFLY_TEAM_ID}")
elseif(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Coinfly Windows target requires AMD64")
  endif()
  add_library(coinfly_vault SHARED windows/credential_bridge.cpp)
  target_link_libraries(coinfly_vault PRIVATE Advapi32 Wintrust Crypt32)
  set(COINFLY_SIGNER_CERT_SHA256_HEX "" CACHE STRING "Reviewed release signer pin; empty denies access")
  target_compile_definitions(coinfly_vault PRIVATE COINFLY_SIGNER_CERT_SHA256_HEX="${COINFLY_SIGNER_CERT_SHA256_HEX}")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    message(FATAL_ERROR "Coinfly Linux desktop target requires AMD64")
  endif()
  # Neural execution is independent of release signing qualification.
  # Linux setup credentials use the explicitly selected desktop Secret Service.
else()
  message(FATAL_ERROR "Desktop targets: macOS ARM64, Windows AMD64, Linux AMD64")
endif()

add_library(coinfly_neural SHARED ../src/coinfly/research/neural/kernel.cpp)
set_target_properties(coinfly_neural PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
install(TARGETS coinfly_neural LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
if(TARGET coinfly_vault)
  install(TARGETS coinfly_vault LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
endif()
# Building a library does not qualify it. Signed-binary hashes and platform
# validation evidence must be reviewed before adding package trust anchors.
